```python
# src/algo/plugins/base.py
from abc import ABC, abstractmethod
import time
import cv2
import numpy as np


class Detector(ABC):
    """每个检测项 = 一个插件：可启停、可排序、可超时、可单独测试"""
    name = "base"
    timeout_ms = 100

    def __init__(self, cfg=None):
        self.cfg = cfg or {}
        self.enabled = self.cfg.get("enabled", True)

    @abstractmethod
    def run(self, frame, ctx):
        ...  # 返回 dict，必须包含 ok(bool)


class BrightnessCheck(Detector):
    name = "brightness"

    def run(self, frame, ctx):
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        mean = float(gray.mean())
        lo, hi = self.cfg.get("min", 40), self.cfg.get("max", 220)
        return {"ok": lo <= mean <= hi, "mean": round(mean, 2), "detail": "曝光检查"}


class BlobDefectCheck(Detector):
    name = "blob_defect"

    def run(self, frame, ctx):
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        blur = cv2.GaussianBlur(gray, (5, 5), 0)
        _, bw = cv2.threshold(blur, 0, 255,
                              cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
        n, labels, stats, cents = cv2.connectedComponentsWithStats(bw, 8)
        min_area = self.cfg.get("min_area", 50)
        boxes = [tuple(map(int, stats[i, :4])) for i in range(1, n)
                 if stats[i, cv2.CC_STAT_AREA] >= min_area]
        return {"ok": len(boxes) == 0, "count": len(boxes),
                "boxes": boxes, "detail": "暗缺陷 Blob"}


REGISTRY = {"brightness": BrightnessCheck, "blob_defect": BlobDefectCheck}


class Pipeline:
    def __init__(self, flow_cfg):
        self.items = []
        for item in flow_cfg:
            cls = REGISTRY.get(item["type"])
            if cls is None:
                raise ValueError(f"未知插件: {item['type']}")
            self.items.append(cls(item.get("params", {})))

    def run(self, frame):
        ctx, results = {}, []
        for p in self.items:
            if not p.enabled:
                results.append({"name": p.name, "skipped": True})
                continue
            t0 = time.perf_counter()
            try:
                r = p.run(frame, ctx)
                r.setdefault("ok", False)
            except Exception as e:                       # 单个插件异常 → 降级为 NG，绝不崩主流程
                r = {"ok": False, "error": repr(e)}
            r["name"] = p.name
            r["cost_ms"] = round((time.perf_counter() - t0) * 1000, 2)
            if r["cost_ms"] > p.timeout_ms:
                r["ok"] = False
                r["timeout"] = True
            results.append(r)
            ctx[p.name] = r
        return {"ok": all(r.get("ok", True) for r in results if not r.get("skipped")),
                "items": results}


if __name__ == "__main__":
    flow = [
        {"type": "brightness", "params": {"min": 40, "max": 220}},
        {"type": "blob_defect", "params": {"min_area": 50}},
    ]
    img = np.full((300, 400, 3), 190, np.uint8)
    cv2.circle(img, (200, 150), 12, 45, -1)             # 故意放一个缺陷
    print(Pipeline(flow).run(img))
```

```python
# src/business/fsm.py
from enum import Enum
import time


class State(Enum):
    IDLE = "IDLE"
    CAPTURE = "CAPTURE"
    ALGO = "ALGO"
    SEND = "SEND"
    WAIT_ACK = "WAIT_ACK"
    NG = "NG"
    ALARM = "ALARM"


class InspectionFSM:
    """事件驱动 + 显式迁移表 + 每状态独立超时"""

    TRANSITIONS = {
        State.IDLE:     {"TRIG": State.CAPTURE},
        State.CAPTURE:  {"FRAME_OK": State.ALGO, "TIMEOUT": State.NG},
        State.ALGO:     {"DONE": State.SEND, "TIMEOUT": State.NG},
        State.SEND:     {"SENT": State.WAIT_ACK, "TIMEOUT": State.ALARM},
        State.WAIT_ACK: {"ACK": State.IDLE, "TIMEOUT": State.ALARM},
        State.NG:       {"RESET": State.IDLE},
        State.ALARM:    {"RECONNECTED": State.IDLE},
    }

    def __init__(self, comm, pipeline, capture, timeout_s=1.0):
        self.state = State.IDLE
        self.comm, self.pipeline, self.capture = comm, pipeline, capture
        self.timeout_s = timeout_s
        self.entered_at = time.monotonic()
        self.result = None

    def on_event(self, event, payload=None):
        nxt = self.TRANSITIONS.get(self.state, {}).get(event)
        if nxt is None:
            print(f"[FSM] 忽略非法事件 {event} @ {self.state.value}")
            return
        print(f"[FSM] {self.state.value} --{event}--> {nxt.value}")
        self.state = nxt
        self.entered_at = time.monotonic()
        self._enter(payload)

    def tick(self):
        """由主循环每 10ms 调用一次，不阻塞、不 sleep"""
        if time.monotonic() - self.entered_at > self.timeout_s:
            self.on_event("TIMEOUT")

    def _enter(self, payload):
        if self.state == State.CAPTURE:
            frame = self.capture()
            self.on_event("FRAME_OK" if frame is not None else "TIMEOUT", frame)
        elif self.state == State.ALGO:
            self.result = self.pipeline(payload)
            self.on_event("DONE", self.result)
        elif self.state == State.SEND:
            ok = self.comm.send("RESULT," + ("OK" if payload["ok"] else "NG"))
            self.on_event("SENT" if ok else "TIMEOUT")
        elif self.state == State.WAIT_ACK:
            self.on_event("ACK")
        elif self.state in (State.NG, State.ALARM):
            self.on_event("RESET" if self.state == State.NG else "RECONNECTED")


if __name__ == "__main__":
    import numpy as np
    import cv2

    class Comm:
        def send(self, msg):
            print("[COMM]", msg)
            return True

    flow = [{"type": "brightness", "params": {"min": 40, "max": 220}}]
    try:
        from src.algo.plugins.base import Pipeline      # 从仓库根目录运行
    except ModuleNotFoundError:
        import sys
        from pathlib import Path
        sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
        from src.algo.plugins.base import Pipeline
    fsm = InspectionFSM(Comm(), Pipeline(flow),
                        lambda: np.full((100, 100, 3), 180, np.uint8))
    fsm.on_event("TRIG")
    print("最终状态:", fsm.state.value)
```

> **C# / 上位机迁移提示**：你工作中的上位机 HT5 是 C# WinForms 程序，任务是"逐条拉取任务、顺序执行"（这里借用你现有项目背景做类比，**没有单独的"项目约定"文件**）。这里的 `tick()` + 显式迁移表就是同一套思路：**不要在 UI 线程里 `Thread.Sleep` 等 PLC 信号**，用状态 + 超时 + 事件驱动，否则界面假死、掉线无法恢复。

```python
# src/capture/async_capture.py
import threading
import time
import queue
import cv2
import numpy as np


class AsyncCapture:
    """采集线程与算法线程分离；队列 maxsize=1 保证算法永远处理最新帧"""

    def __init__(self, source=0):
        self.cap = cv2.VideoCapture(source)
        self.q = queue.Queue(maxsize=1)
        self.stopped = False
        self.fps = 0.0
        self.thread = threading.Thread(target=self._loop, daemon=True)

    def _loop(self):
        n, t0 = 0, time.perf_counter()
        while not self.stopped:
            ok, frame = self.cap.read()
            if not ok:
                self.stopped = True
                break
            if self.q.full():
                try:
                    self.q.get_nowait()      # 丢旧帧，绝不积压
                except queue.Empty:
                    pass
            self.q.put(frame)
            n += 1
            dt = time.perf_counter() - t0
            if dt >= 1.0:
                self.fps, n, t0 = n / dt, 0, time.perf_counter()
        self.cap.release()

    def start(self):
        self.thread.start()
        return self

    def read(self, timeout=0.5):
        try:
            return self.q.get(timeout=timeout)
        except queue.Empty:
            return None

    def stop(self):
        self.stopped = True
        self.thread.join(timeout=1.0)


if __name__ == "__main__":
    # 无相机时用合成帧演示接口（真实项目换成 AsyncCapture(0).start()）
    cap = None
    frame = np.full((240, 320, 3), 128, np.uint8)
    for i in range(3):
        t0 = time.perf_counter()
        _ = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        print(f"帧 {i}: 算法耗时 {(time.perf_counter()-t0)*1000:.2f} ms")
    print("结论：采集与算法解耦后，算法慢不会拖累采集，也不会积压旧帧")
```

**优化顺序**（原文 `:354-357`、`:497`）：ROI → 降采样粗定位 → LUT/向量化 → 多线程/多进程 → 最后才考虑换硬件。产线 100ms 节拍，算法跑 2 秒，再准也没用（坑 6 `:750-751`）。

```python
class ReconnectClient:
    """断线自动重连 + 心跳：一个异常不能把整条线停了（:482-483）"""

    def __init__(self, connect_fn, heartbeat_s=2.0, retry_s=1.0):
        self.connect_fn, self.conn = connect_fn, None
        self.heartbeat_s, self.retry_s = heartbeat_s, retry_s
        self.last_hb = 0.0

    def ensure(self):
        if self.conn is None:
            try:
                self.conn = self.connect_fn()
                log.info("通信已连接")
            except OSError as e:
                log.warning(f"重连失败: {e}")
        return self.conn

    def heartbeat(self):
        now = time.monotonic()
        if now - self.last_hb >= self.heartbeat_s:
            self.last_hb = now
            if not self.ensure():
                log.warning("心跳丢失，等待重连")
                return False
        return True
```

```python
    costs.sort()

          f"P95 {np.percentile(costs, 95):.2f}ms，最大 {costs[-1]:.2f}ms")
    print("8 小时压测时同步盯任务管理器：内存曲线应平坦（:245、:620）")


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--seconds", type=int, default=60)
    stress(ap.parse_args().seconds)
```

## 11. Q&A 交叉引用 · `AsyncCapture` 里的 `ok`（Q07）

§3 `:293-296` 写的是 `ok, frame = self.cap.read()` 紧跟 `if not ok: self.stopped = True; break`。这里的 **`ok` 就是 `ret`**：`cap.read()` 返回 `(retval, image)`，`retval` 表示本次是否拿到一帧非空图像；`ok=False` 时 `frame` 是 `None` 而非空数组。完整语义、C++ 实现判据（`cap.cpp:551` 返回 `!image.empty()`）与 5.0.0 实测见 [`10` 号 §9](./10_速查手册_API_面试题_踩坑_Java类比.md)。

**顺带发现的一个真实隐患**：这段代码遇到**一次** `False` 就永久停线程，且不尝试重连。工业相机的瞬时坏帧 / 短暂掉线完全可能只失败一两帧，按 W41–44 的 8 小时压测要求，这会让采集静默停摆而算法线程还在等队列。另外实测 EOF 后 `cap.set(CAP_PROP_POS_FRAMES, 0)` 返回 `True` 也读不回来，必须 `release()` + `open()` 重开。

建议改成：连续 `N` 帧失败才判掉线 → 重新 `open()` → 仍失败则通知上位机 / PLC 停机并记日志。可直接参考 `10` §9.5 的 B 模板。**本节只做交叉引用，不改 §3 原代码**。
