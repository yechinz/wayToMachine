## 1. 相机模型与畸变（S-31）

针孔模型：`s · [u, v, 1]ᵀ = K · [R|t] · [X, Y, Z, 1]ᵀ`

```
K = [ fx  0  cx ]      fx, fy：焦距（像素单位）    cx, cy：主点
    [ 0  fy  cy ]      dist = [k1, k2, p1, p2, k3]
    [ 0   0   1 ]      k*：径向畸变（桶形/枕形）   p*：切向畸变（镜头装歪）
```

| 名词 | 含义 | 现场影响 |
|---|---|---|
| 内参 K | 相机自身属性 | 换镜头/调焦后必须重标 |
| 畸变系数 | 镜头与装配误差 | 视野边缘直线变弯、尺寸偏差最大 |
| 外参 R/t | 相机相对标定板的姿态 | 每张标定图都不同，但**不是我们要的** |
| 重投影误差 | 用求出的参数把 3D 点投回图像，与实测角点的平均偏差 | **< 0.5 px 算好，> 1 px 说明板不平/图模糊**（`:434`） |

## 2. 张正友标定：完整可运行代码（T0-05 / T1-17）

### 2.1 无相机也能跑通：模拟 20 张不同姿态标定图

```python
# src/algo/calibrate.py —— 支持"模拟棋盘图"和"真实照片"两种输入
import glob
import json
from pathlib import Path
import cv2
import numpy as np

PATTERN = (9, 6)        # 内角点数（列, 行）
SQUARE_MM = 10.0        # 每个方格真实边长，单位 mm（打印后实测！）


def make_board(cols=10, rows=7, sq=40, canvas=(640, 480)):
    """生成一张棋盘格图（10x7 个方格 → 9x6 个内角点）"""
    img = np.full((canvas[1], canvas[0]), 255, np.uint8)
    x0 = (canvas[0] - cols * sq) // 2
    y0 = (canvas[1] - rows * sq) // 2
    for r in range(rows):
        for c in range(cols):
            if (r + c) % 2 == 0:
                cv2.rectangle(img,
                              (x0 + c * sq, y0 + r * sq),
                              (x0 + (c + 1) * sq - 1, y0 + (r + 1) * sq - 1),
                              0, -1)
    return img


def simulate_views(n=20, out_dir="data/calib_sim"):
    """用旋转+缩放+平移模拟 20 张不同姿态（真实项目请换成相机实拍）"""
    Path(out_dir).mkdir(parents=True, exist_ok=True)
    board = make_board()
    h, w = board.shape
    rng = np.random.default_rng(7)
    for i in range(n):
        ang = float(rng.uniform(-12, 12))
        scale = float(rng.uniform(0.88, 1.05))
        M = cv2.getRotationMatrix2D((w / 2, h / 2), ang, scale)
        M[0, 2] += float(rng.uniform(-25, 25))
        M[1, 2] += float(rng.uniform(-20, 20))
        view = cv2.warpAffine(board, M, (w, h), borderValue=255)
        cv2.imwrite(f"{out_dir}/{i:02d}.png", view)
    print(f"已生成 {n} 张模拟标定图 → {out_dir}")


def calibrate(image_glob, pattern=PATTERN, square_mm=SQUARE_MM):
    objp = np.zeros((pattern[0] * pattern[1], 3), np.float32)
    objp[:, :2] = np.mgrid[0:pattern[0], 0:pattern[1]].T.reshape(-1, 2) * square_mm

    obj_pts, img_pts, used = [], [], []
    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 1e-3)
    for f in sorted(glob.glob(image_glob)):
        raw = np.fromfile(f, dtype=np.uint8)          # 兼容中文路径
        img = cv2.imdecode(raw, cv2.IMREAD_COLOR)
        if img is None:
            continue
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        ok, corners = cv2.findChessboardCorners(gray, pattern, None)
        if not ok:
            continue
        corners = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)
        obj_pts.append(objp)
        img_pts.append(corners)
        used.append(f)

    if len(obj_pts) < 8:
        raise RuntimeError(f"只有 {len(obj_pts)} 张有效，需 ≥ 8 张（推荐 15–30 张，:420）")

    rms, K, dist, rvecs, tvecs = cv2.calibrateCamera(
        obj_pts, img_pts, gray.shape[::-1], None, None)
    print(f"有效图 {len(obj_pts)} 张，重投影误差 RMS = {rms:.3f} px（<0.5 算好）")
    print("K =\n", np.round(K, 3))
    print("dist =", np.round(dist.ravel(), 5))
    result = dict(rms=float(rms), K=K.tolist(), dist=dist.ravel().tolist(),
                  image_size=[int(gray.shape[1]), int(gray.shape[0])],
                  square_mm=square_mm, pattern=list(pattern), used=used)
    Path("configs").mkdir(exist_ok=True)
    Path("configs/camera.yaml.json").write_text(
        json.dumps(result, ensure_ascii=False, indent=2), encoding="utf-8")
    return result


if __name__ == "__main__":
    simulate_views()
    calibrate("data/calib_sim/*.png")
    # 真实项目：把 calibrate() 的参数换成 "data/calib/*.jpg" 即可，流程一行不改
```

> **模拟数据的诚实说明**：合成图不含真实镜头畸变，所以 `dist` 接近 0、`rms` 很小。这段代码的价值是**把流程跑通并看懂每个返回值**；真实相机必须打印棋盘格（或买标定板）、拍 15–30 张**不同姿态且覆盖全视野**的照片（`:420`、`:429`）。

### 2.2 实拍 15–30 张的质量要点（原文 `:429-434`）

| 要点 | 做错会怎样 |
|---|---|
| 姿态多样（倾斜 ±30°、旋转、远近） | 全是正视图 → 焦距与畸变解耦失败 |
| 覆盖视野四角与中心 | 边缘畸变外推，边缘测量误差大 |
| 标定板平整、贴牢，不反光 | 重投影误差 > 1 px |
| 图像清晰、无运动模糊 | `cornerSubPix` 收敛到错误位置 |
| 打印方格的**实际边长**要量，不能信 PDF 标称 | 像素当量整体比例错 |

## 3. 畸变校正与三种几何变换（S-29、S-30、S-32）

```python
import cv2
import numpy as np
import json
from pathlib import Path


def undistort_fast(img, calib):
    """实时视频用：预生成映射表，每帧只做 remap"""
    K = np.array(calib["K"], np.float64)
    dist = np.array(calib["dist"], np.float64)
    w, h = calib["image_size"]
    map1, map2 = cv2.initUndistortRectifyMap(K, dist, None, K, (w, h), cv2.CV_16SC2)
    return cv2.remap(img, map1, map2, cv2.INTER_LINEAR)


def undistort_simple(img, calib):
    """单张图用：一行搞定，但每帧都重算映射，慢"""
    K = np.array(calib["K"], np.float64)
    dist = np.array(calib["dist"], np.float64)
    return cv2.undistort(img, K, dist)


def geometry_demo(img):
    h, w = img.shape[:2]
    # ① 仿射：平移+旋转+缩放（3 个点确定）
    M = cv2.getRotationMatrix2D((w / 2, h / 2), 30, 0.9)
    aff = cv2.warpAffine(img, M, (w, h), borderValue=(255, 255, 255))

    # ② 透视：4 个点确定，能表达"近大远小"（9 点标定之后常用）
    src = np.float32([[0, 0], [w - 1, 0], [w - 1, h - 1], [0, h - 1]])
    dst = np.float32([[30, 20], [w - 60, 50], [w - 20, h - 30], [50, h - 60]])
    H = cv2.getPerspectiveTransform(src, dst)
    persp = cv2.warpPerspective(img, H, (w, h), borderValue=(255, 255, 255))

    # ③ 极坐标展开：环形件（瓶盖/轴承/O 型圈）拉成矩形，字符/缺陷好检测
    polar = cv2.warpPolar(img, (360, 120), (w / 2, h / 2), min(w, h) / 2,
                          cv2.INTER_LINEAR + cv2.WARP_POLAR_LINEAR)
    return aff, persp, polar


if __name__ == "__main__":
    img = np.full((240, 320, 3), 240, np.uint8)
    cv2.circle(img, (160, 120), 60, (30, 30, 30), 2)
    a, p, pol = geometry_demo(img)
    print("仿射/透视/极坐标输出尺寸:", a.shape, p.shape, pol.shape)
    print("极坐标展开后: 行=半径方向, 列=角度方向，环形字符被拉直")
```

| 变换 | 自由度 | 能表达 | 不能表达 |
|---|---|---|---|
| 仿射 `warpAffine` | 6 | 平移、旋转、缩放、剪切 | 透视（近大远小）、镜头畸变 |
| 透视 `warpPerspective` | 8 | 以上 + 透视 | 非线性畸变（需 remap） |
| `remap` + 映射表 | 任意 | 任意像素重映射（含畸变） | — |

> **九点标定的边界**（`:432`）：仿射解决不了镜头畸变和 **Z 轴高度变化**。工件高度一变，倍率就变，必须做畸变校正 + 保持工作面高度恒定，或上远心镜头。

## 4. 像素当量与标准件标定（S-33）

**理论值**（选型阶段用，`:273`）：`mm/px = FOV宽度(mm) / 相机水平像素数`。
**实际值**（验收阶段必须用）：拿已知尺寸的标准件（量块/硬币/圆点靶）拍图，量出它的像素尺寸，再除。

```python
import cv2
import numpy as np


def pixel_equivalent(gray, known_diameter_mm):
    """用已知直径的圆标准件求像素当量 mm/px"""
    _, bw = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    cnts = cv2.findContours(bw, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)[-2]
    c = max(cnts, key=cv2.contourArea)
    (_, _), radius_px = cv2.minEnclosingCircle(c)
    mm_per_px = known_diameter_mm / (2 * radius_px)
    print(f"标准件直径 {2*radius_px:.2f} px = {known_diameter_mm} mm "
          f"→ 像素当量 {mm_per_px:.6f} mm/px")
    return mm_per_px


if __name__ == "__main__":
    g = np.full((400, 400), 30, np.uint8)
    cv2.circle(g, (200, 200), 100, 220, -1, cv2.LINE_AA)
    mmpx = pixel_equivalent(g, 25.0)     # 假设标准件直径 25mm
    print("理论值 100mm/2448px ≈", round(100 / 2448, 6), "mm/px（:281）")
```

> **为什么不能用理论值交差**：镜头实际焦距有公差、工作距离装不准、镜头有畸变、靶面尺寸标称有偏差。**理论值用于选型，标准件实测值用于出报告**（`:387`）。

## 5. 亚像素边缘与拟合（S-34、S-35）

### 5.1 卡尺工具：沿法线找梯度极值 + 抛物线插值

整数像素边缘只能给到 ±1 px，按 0.041 mm/px 算就是 ±0.04mm——**单这一项就吃光了 0.05mm 的指标**。所以必须亚像素：

```python
import cv2
import numpy as np


def refine_peak(profile, i):
    """抛物线插值：用峰值及左右两点拟合 y=a(x-d)²+b，返回亚像素峰位"""
    if i <= 0 or i >= len(profile) - 1:
        return float(i)
    y0, y1, y2 = (float(v) for v in profile[i - 1:i + 2])
    denom = y0 - 2 * y1 + y2
    if abs(denom) < 1e-9:
        return float(i)
    delta = 0.5 * (y0 - y2) / denom
    return float(i) + float(np.clip(delta, -1.0, 1.0))


def caliper_diameter(gray, approx_cx, approx_cy, mm_per_px):
    """沿过圆心的水平扫描线，找左右两侧梯度峰并做亚像素细化"""
    row = gray[int(approx_cy), :].astype(np.float32)
    row = cv2.GaussianBlur(row.reshape(1, -1), (1, 5), 0).ravel()
    grad = np.abs(np.diff(row))
    half = len(grad) // 2
    left_i = int(np.argmax(grad[:half]))
    right_i = half + int(np.argmax(grad[half:]))
    left = refine_peak(grad, left_i) + 0.5        # +0.5 补偿 diff 的偏移
    right = refine_peak(grad, right_i) + 0.5
    diameter_px = right - left
    return diameter_px, diameter_px * mm_per_px
```

## 6. P2 完整项目：尺寸测量 + GR&R（W7，T1-26）

```python
# src/algo/p2_measure.py —— 全流程可运行，输出 GR&R 数据
import cv2
import numpy as np
import json
from pathlib import Path

MM_PER_PX = 0.041        # 合成图演示用的理论值（路线图 :281）；实拍必须换成 §4 标准件标定的实测值，否则误差结论不成立
TRUE_DIAM_MM = 25.0
RADIUS_PX = TRUE_DIAM_MM / MM_PER_PX / 2     # 反推合成图里的圆半径


def make_workpiece(cx=320, drift=1.0, noise_sigma=2.0, seed=0):
    """合成一个'标准件圆'；drift 模拟视野边缘的透视倍率变化"""
    h, w = 480, 640
    img = np.full((h, w), 40, np.uint8)
    r = RADIUS_PX * drift
    cv2.circle(img, (int(cx), h // 2), int(round(r)), 210, -1, cv2.LINE_AA)
    rng = np.random.default_rng(seed)
    img = np.clip(img.astype(np.float32) + rng.normal(0, noise_sigma, img.shape),
                  0, 255).astype(np.uint8)
    return img, h // 2


def refine_peak(profile, i):
    if i <= 0 or i >= len(profile) - 1:
        return float(i)
    y0, y1, y2 = (float(v) for v in profile[i - 1:i + 2])
    den = y0 - 2 * y1 + y2
    if abs(den) < 1e-9:
        return float(i)
    return float(i) + float(np.clip(0.5 * (y0 - y2) / den, -1.0, 1.0))


def measure(img, cy, mm_per_px=MM_PER_PX):
    row = img[cy, :].astype(np.float32)
    row = cv2.GaussianBlur(row.reshape(1, -1), (1, 5), 0).ravel()
    grad = np.abs(np.diff(row))
    cx = len(grad) // 2
    li = int(np.argmax(grad[:cx]))
    ri = cx + int(np.argmax(grad[cx:]))
    left = refine_peak(grad, li) + 0.5
    right = refine_peak(grad, ri) + 0.5
    return (right - left) * mm_per_px


def grr_report(values, true_mm, tolerance_mm=0.05):
    v = np.array(values, float)
    err = v - true_mm
    rep = {
        "n": len(v),
        "mean_mm": round(float(v.mean()), 5),
        "std_mm": round(float(v.std(ddof=1)), 5),
        "range_mm": round(float(v.max() - v.min()), 5),
        "bias_mm": round(float(err.mean()), 5),
        "max_abs_error_mm": round(float(np.abs(err).max()), 5),
        "grr_ratio_vs_tolerance": round(float(v.std(ddof=1) / tolerance_mm), 3),
    }
    print(json.dumps(rep, ensure_ascii=False, indent=2))
    return rep


if __name__ == "__main__":
    # ① 重复性：同一工件测 30 次（每次传感器噪声不同）
    vals = []
    for i in range(30):
        img, cy = make_workpiece(seed=i)
        vals.append(measure(img, cy))
    grr_report(vals, TRUE_DIAM_MM)

    # ② 误差随视野位置变化：模拟未做畸变/透视校正时边缘倍率漂移
    print("\n--- 误差 vs 视野位置（故意保留倍率漂移，观察退化）---")
    curve = []
    for x in range(120, 521, 50):
        drift = 1.0 + 0.0006 * (x - 320)        # 边缘略放大，模拟透视误差
        img, cy = make_workpiece(cx=x, drift=drift, noise_sigma=1.0, seed=99)
        d = measure(img, cy)
        curve.append((x, round(float(d - TRUE_DIAM_MM), 4)))
        print(f"x={x:3d}px  误差 {d - TRUE_DIAM_MM:+.4f} mm")
    print("曲线数据:", curve)
    print("结论：边缘误差更大 → 必须先畸变校正 + 覆盖全视野的标定（:429、:539）")
```

**P2 交付物检查表**（原文 `:537-540`）：

- [ ] 直径误差 < 0.05mm（用 500 万像素 + 普通镜头、FOV 100mm）
- [ ] 误差-视野位置曲线 + 原因解释（畸变 / 透视）
- [ ] 一份 GR&R 报告（均值、标准差、极差、标准差占公差带比例 ≤ 10%，`:388`）
- [ ] 30 次重复性数据 + 换光照/角度后的退化记录
- [ ] 简历句：`基于张正友标定与亚像素边缘拟合实现零件尺寸测量，重复性标准差 0.02mm，测量节拍 80ms/件。`

## 7. 误差来源清单（面试问"你的精度凭什么可信"就答这个）

| 来源 | 量级 | 对策 |
|---|---|---|
| 镜头畸变 | 边缘最大 | 内参标定 + remap 校正 |
| 像素当量用理论值 | 整体比例偏差 | 标准件实测（§4） |
| 整数像素边缘 | ±0.04mm 量级 | 亚像素插值（§5.1） |
| 工件高度变化 / 不水平 | 倍率变化 | 远心镜头、夹具定位 |
| 温漂（长时间运行） | 缓变 | 定期用标准件复位（`:389`） |
| 打光导致的边缘灰度漂移 | 随机 | 背光 + 固定曝光 |
| 阈值/拟合参数漂移 | 随机 | 回归图库 + 参数版本 |
