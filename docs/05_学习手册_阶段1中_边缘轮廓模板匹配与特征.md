## 0. 本章地图

```
04 号手册的干净二值图
   ├─ W3  边缘（Canny 五步） → 轮廓（RETR 模式） → 几何特征 → 计数/编号
```

**一句话提纲**：边缘是"灰度突变的地方"，轮廓是"把这些地方连成一圈"，模板/特征解决的是"工件每次放的位置不一样"。

> **Java 类比**：`findContours` ≈ 对二值图做一次 flood fill / 连通域标记后把边界链码抽出来；`RETR_TREE` ≈ 保留父子嵌套关系的树结构（外轮廓是父，孔是子）；`matchTemplate` ≈ 一个双重循环里算滑动窗口的相似度，返回一张"得分图"。

## 1. 梯度与 Canny 五步（S-19、S-20）

### 1.1 三种梯度算子的差别

```python
import cv2
import numpy as np

# 合成：白底黑圆 + 细线，带轻微噪声
img = np.full((300, 400), 235, np.uint8)
cv2.circle(img, (150, 150), 60, 60, -1)
cv2.line(img, (250, 60), (370, 250), 100, 2)
rng = np.random.default_rng(0)
img = np.clip(img.astype(np.float32) + rng.normal(0, 6, img.shape),
              0, 255).astype(np.uint8)

blur = cv2.GaussianBlur(img, (5, 5), 0)
gx = cv2.Sobel(blur, cv2.CV_32F, 1, 0, ksize=3)
gy = cv2.Sobel(blur, cv2.CV_32F, 0, 1, ksize=3)
mag, ang = cv2.cartToPolar(gx, gy, angleInDegrees=True)
print("Sobel 幅值范围:", round(float(mag.min()), 2), round(float(mag.max()), 2))

sx = cv2.Scharr(blur, cv2.CV_32F, 1, 0)
lap = cv2.Laplacian(blur, cv2.CV_32F, ksize=3)
print("Scharr 对小核更精确；Laplacian 是二阶导，对噪声最敏感:",
      round(float(np.abs(lap).mean()), 2))
```

| 算子 | 阶数 | 特点 |
|---|---|---|
| Sobel | 一阶 + 平滑 | 通用，工业默认 |
| Scharr | 一阶 | 3×3 核旋转对称性更好，细边更准 |
| Laplacian | 二阶 | 对噪声极敏感，常用于"清晰度评价"而非直接提边 |

### 1.2 手写 Canny 五步（能默写出这五步，面试题 16 就稳了）

```
① 高斯滤波去噪
② 计算梯度幅值与方向（Sobel）
③ 非极大值抑制 NMS：沿梯度方向只保留局部最大 → 边缘变细到 1 像素
④ 双阈值：>high 强边缘，low~high 弱边缘
⑤ 滞后连接：弱边缘只有与强边缘相连才保留
```

```python
import cv2
import numpy as np


def canny_manual(gray, low=50, high=150):
    # ① 高斯
    blur = cv2.GaussianBlur(gray, (5, 5), 0)
    # ② 梯度
    gx = cv2.Sobel(blur, cv2.CV_32F, 1, 0, ksize=3)
    gy = cv2.Sobel(blur, cv2.CV_32F, 0, 1, ksize=3)
    mag, ang = cv2.cartToPolar(gx, gy, angleInDegrees=True)

    # ③ NMS：把方向量化为 0°/45°/90°/135°，比较正负方向邻居
    q = np.zeros(ang.shape, np.uint8)
    q[(ang >= 22.5) & (ang < 67.5)] = 1      # 45°
    q[(ang >= 67.5) & (ang < 112.5)] = 2     # 90°
    q[(ang >= 112.5) & (ang < 157.5)] = 3    # 135°
    # 其余（含 157.5–180 与 0–22.5）归 0°
    nms = np.zeros_like(mag)
    shifts = {0: (0, 1), 1: (1, 1), 2: (1, 0), 3: (1, -1)}
    for d, (dy, dx) in shifts.items():
        m = q == d
        a = np.roll(mag, (-dy, -dx), axis=(0, 1))
        b = np.roll(mag, (dy, dx), axis=(0, 1))
        nms[m] = mag[m] * ((mag[m] >= a[m]) & (mag[m] >= b[m]))

    # ④⑤ 双阈值 + 滞后连接（迭代膨胀近似）
    strong = (nms >= high).astype(np.uint8)
    weak = ((nms >= low) & (nms < high)).astype(np.uint8)
    edge = strong.copy()
    kernel = np.ones((3, 3), np.uint8)
    for _ in range(6):
        grown = cv2.dilate(edge, kernel)
        edge = np.maximum(edge, grown & weak)
    return (edge * 255).astype(np.uint8)


if __name__ == "__main__":
    img = np.full((200, 300), 230, np.uint8)
    cv2.circle(img, (150, 100), 45, 70, -1)
    mine = canny_manual(img, 40, 120)
    official = cv2.Canny(cv2.GaussianBlur(img, (5, 5), 0), 40, 120)
    print("手写边缘像素:", int((mine > 0).sum()),
          " 官方 Canny:", int((official > 0).sum()))
    print("两者形态应接近；手写版用于理解，产线一律用 cv2.Canny")
```

> **注意**：手写版是为了理解五步，产线用 `cv2.Canny`（C++ 优化、细节更完整）。**双阈值经验比 `high:low ≈ 2:1 ~ 3:1`**；调参时先固定 low，只调 high。

## 2. 轮廓：检索模式与近似方法（S-21、S-22、S-23）

### 2.1 `RETR_EXTERNAL` vs `RETR_TREE`（阶段 1 验收题 `:184`）

```python
import cv2
import numpy as np

# 合成：一个大圆里面套一个"孔"，模拟带孔零件
bw = np.zeros((220, 300), np.uint8)
cv2.circle(bw, (150, 110), 80, 255, -1)
cv2.circle(bw, (150, 110), 30, 0, -1)

for mode, name in [(cv2.RETR_EXTERNAL, "RETR_EXTERNAL"),
                   (cv2.RETR_TREE, "RETR_TREE"),
                   (cv2.RETR_CCOMP, "RETR_CCOMP")]:
    cnts = cv2.findContours(bw, mode, cv2.CHAIN_APPROX_SIMPLE)[-2]
    areas = sorted(int(cv2.contourArea(c)) for c in cnts)
    print(f"{name:14s} 轮廓数={len(cnts)} 面积={areas}")
```

| 模式 | 返回 | 适用 |
|---|---|---|
| `RETR_EXTERNAL` | **只取最外圈** | 数零件、找外轮廓尺寸（最常用） |
| `RETR_TREE` | 全部 + 父子层级 | 数孔、内外轮廓都要、嵌套结构 |
| `RETR_CCOMP` | 两层（外/内） | 只要"孔"的层级 |
| `CHAIN_APPROX_SIMPLE` | 只存折线端点 | 省内存，几何计算够用 |
| `CHAIN_APPROX_NONE` | 存全部边界点 | 需要逐点分析时 |

## 3. W3 交付：硬币/零件计数与编号（T1-11）

```python
# src/algo/count_parts.py —— 合成图可直接跑；换真图只需改 make_demo()
import cv2
import numpy as np


def make_demo():
    """5 个不同大小的圆 + 1 个矩形 + 噪声点，模拟料盘里的零件"""
    img = np.full((420, 560, 3), 35, np.uint8)       # 深色背景
    for cx, cy, r in [(90, 100, 42), (230, 90, 38), (380, 120, 46),
                      (150, 280, 34), (420, 300, 40)]:
        cv2.circle(img, (cx, cy), r, (190, 190, 195), -1, cv2.LINE_AA)
        cv2.circle(img, (cx, cy), r - 8, (150, 150, 158), 2, cv2.LINE_AA)
    cv2.rectangle(img, (270, 220), (340, 270), (185, 185, 190), -1)
    rng = np.random.default_rng(2)
    for _ in range(12):
        x, y = rng.integers(0, 560), rng.integers(0, 420)
        cv2.circle(img, (int(x), int(y)), 3, (170, 170, 170), -1)
    return img


def count_parts(bgr, min_area=800):
    gray = cv2.cvtColor(bgr, cv2.COLOR_BGR2GRAY)
    blur = cv2.GaussianBlur(gray, (5, 5), 0)
    _, bw = cv2.threshold(blur, 0, 255,
                          cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    bw = cv2.morphologyEx(bw, cv2.MORPH_OPEN, k)      # 去噪点
    bw = cv2.morphologyEx(bw, cv2.MORPH_CLOSE, k)     # 补小孔

    out = bgr.copy()
    cnts = cv2.findContours(bw, cv2.RETR_EXTERNAL,
                            cv2.CHAIN_APPROX_SIMPLE)[-2]
    rows = []
    for i, c in enumerate(sorted(cnts, key=lambda c: cv2.boundingRect(c)[0]), 1):
        area = cv2.contourArea(c)
        if area < min_area:
            continue
        peri = cv2.arcLength(c, True)
        (cx, cy), (rw, rh), _ = cv2.minAreaRect(c)
        circ = 4 * np.pi * area / (peri ** 2) if peri > 0 else 0     # 圆度
        box = cv2.boxPoints(cv2.minAreaRect(c)).astype(np.int32)
        cv2.drawContours(out, [box], 0, (0, 255, 0), 2)
        cv2.putText(out, f"#{i}", (int(cx) - 12, int(cy)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)
        rows.append(dict(id=i, area=round(area, 1), peri=round(peri, 1),
                         w=round(rw, 1), h=round(rh, 1),
                         circularity=round(float(circ), 3),
                         center=(round(cx, 1), round(cy, 1))))
    return out, rows


if __name__ == "__main__":
    img = make_demo()
    vis, rows = count_parts(img)
    for r in rows:
        print(r)
    print("合格零件数:", len(rows))
    cv2.imwrite("docs/parts_count.png", vis)
```

**验收动作**：把 `min_area` 从 800 改成 100，观察噪声点被算进零件；再改成 5000，观察小零件被吃掉。**能解释这两个现象，才叫"会调参"**（原文 `:188`）。

## 4. 模板匹配：NCC 原理与 6 种方法（S-24）

### 4.1 原理（面试题 15 `:831`）

`TM_CCOEFF_NORMED` 本质是**归一化互相关 NCC**：把模板与图像窗口都减去各自均值再算余弦相似度，因此对**整体亮度线性变化**不敏感：

```
R(x,y) = Σ (T' · I') / sqrt(Σ T'² · Σ I'²)      T' = T - mean(T), I' = I_window - mean(I_window)
```

| 方法族 | 代表 | 极值 | 特点 |
|---|---|---|---|
| 平方差 | `TM_SQDIFF` / `SQDIFF_NORMED` | **最小值**最佳 | 对亮度变化敏感 |
| 相关 | `TM_CCORR` / `CCORR_NORMED` | 最大值最佳 | 对亮度偏移敏感 |
| 去均值相关 | `TM_CCOEFF` / `CCOEFF_NORMED` | 最大值最佳 | **工业首选，抗亮度变化** |

```python
import cv2
import numpy as np


def make_scene():
    scene = np.full((360, 480), 200, np.uint8)
    cv2.rectangle(scene, (60, 60), (200, 160), 90, -1)      # 目标：深色矩形
    cv2.circle(scene, (120, 110), 18, 210, -1)
    cv2.rectangle(scene, (300, 200), (440, 300), 90, -1)    # 干扰块
    return scene


def match_all(scene, templ):
    for name in ["TM_SQDIFF_NORMED", "TM_CCORR_NORMED", "TM_CCOEFF_NORMED"]:
        method = getattr(cv2, name)
        res = cv2.matchTemplate(scene, templ, method)
        mn, mx, mnl, mxl = cv2.minMaxLoc(res)
        best = mnl if "SQDIFF" in name else mxl
        score = mn if "SQDIFF" in name else mx
        print(f"{name:18s} score={score:.4f} loc={best}")


if __name__ == "__main__":
    s = make_scene()
    t = s[60:160, 60:200].copy()
    match_all(s, t)

    # 破坏性实验：亮度整体 +40，看谁稳
    bright = cv2.add(s, 40)
    print("--- 整体变亮 +40 ---")
    match_all(bright, t)

    # 破坏性实验：模板旋转 15°，NCC 得分坍塌
    M = cv2.getRotationMatrix2D((t.shape[1] / 2, t.shape[0] / 2), 15, 1.0)
    rot = cv2.warpAffine(t, M, (t.shape[1], t.shape[0]), borderValue=200)
    res = cv2.matchTemplate(s, rot, cv2.TM_CCOEFF_NORMED)
    print("--- 模板旋转 15° ---  最佳得分:", round(float(cv2.minMaxLoc(res)[1]), 4))
```

**结论（写进笔记）**：模板匹配只适合**刚性、纹理稳定、无旋转缩放、光照可控**的场景；一旦有旋转/缩放/透视，就要走特征点 + 单应性（§5）或形状匹配（HALCON 强项，`:370`）。

## 5. 特征点 + 单应性：解决大位移与透视（S-25、S-26）

```python
# src/algo/align_by_feature.py
import cv2
import numpy as np


def make_pair():
    """基准图 + 一张旋转/平移/透视变换后的'待对齐图'"""
    base = np.full((320, 420, 3), 245, np.uint8)
    cv2.putText(base, "LOT 2026-A", (40, 90),
                cv2.FONT_HERSHEY_SIMPLEX, 1.1, (30, 30, 30), 3)
    cv2.rectangle(base, (40, 130), (380, 280), (60, 60, 60), 2)
    for x in range(60, 380, 40):
        cv2.circle(base, (x, 205), 8, (40, 40, 40), -1)
    src = np.float32([[0, 0], [419, 0], [419, 319], [0, 319]])
    dst = np.float32([[25, 18], [400, 45], [385, 305], [10, 285]])
    H_true = cv2.getPerspectiveTransform(src, dst)
    moved = cv2.warpPerspective(base, H_true, (420, 320), borderValue=(245, 245, 245))
    # 模拟一处印刷缺陷（缺了一个圆点）
    cv2.circle(moved, (180, 200), 12, (245, 245, 245), -1)
    return base, moved


def make_feature():
    """5.0 主包已自带 SIFT；保留 ORB 回退只为兼容 4.x / headless 环境"""
    if hasattr(cv2, "SIFT_create"):
        return cv2.SIFT_create(), cv2.NORM_L2
    return cv2.ORB_create(nfeatures=1000), cv2.NORM_HAMMING


def align(ref, cur):
    g1 = cv2.cvtColor(ref, cv2.COLOR_BGR2GRAY)
    g2 = cv2.cvtColor(cur, cv2.COLOR_BGR2GRAY)
    feat, norm = make_feature()
    k1, d1 = feat.detectAndCompute(g1, None)
    k2, d2 = feat.detectAndCompute(g2, None)
    if d1 is None or d2 is None or len(k1) < 4 or len(k2) < 4:
        raise RuntimeError("特征点不足，检查图像纹理/光照")

    if norm == cv2.NORM_L2:
        matcher = cv2.BFMatcher(cv2.NORM_L2)
        raw = matcher.knnMatch(d1, d2, k=2)
        # 描述子不足时某个查询可能只返回 1 个邻居，先做长度保护再 Lowe 比值筛选
        good = [m for pair in raw if len(pair) == 2
                for m, n in [pair] if m.distance < 0.75 * n.distance]
    else:
        matcher = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
        good = sorted(matcher.match(d1, d2), key=lambda m: m.distance)[:80]

    if len(good) < 4:
        raise RuntimeError(f"好匹配只有 {len(good)} 个，无法求单应性")
    p1 = np.float32([k1[m.queryIdx].pt for m in good]).reshape(-1, 1, 2)
    p2 = np.float32([k2[m.trainIdx].pt for m in good]).reshape(-1, 1, 2)
    H, mask = cv2.findHomography(p2, p1, cv2.RANSAC, 3.0)   # 待对齐 → 基准
    if H is None:
        raise RuntimeError("单应性求解失败")
    warped = cv2.warpPerspective(cur, H, (ref.shape[1], ref.shape[0]))
    return warped, H, int(mask.sum()), len(good)


def diff_defects(ref, aligned, min_area=40):
    """对齐后差分：W11 的 P3 传统路线之一（:396）"""
    g1 = cv2.cvtColor(ref, cv2.COLOR_BGR2GRAY)
    g2 = cv2.cvtColor(aligned, cv2.COLOR_BGR2GRAY)
    d = cv2.absdiff(g1, g2)
    d = cv2.GaussianBlur(d, (5, 5), 0)
    _, bw = cv2.threshold(d, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    bw = cv2.morphologyEx(bw, cv2.MORPH_OPEN,
                          cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
    n, labels, stats, cents = cv2.connectedComponentsWithStats(bw, 8)
    out = aligned.copy()
    boxes = []
    for i in range(1, n):
        x, y, w, h, area = stats[i]
        if area >= min_area:
            boxes.append((x, y, w, h, int(area)))
            cv2.rectangle(out, (x, y), (x + w, y + h), (0, 0, 255), 2)
    return out, boxes, d


if __name__ == "__main__":
    ref, cur = make_pair()
    aligned, H, inliers, total = align(ref, cur)
    print(f"匹配 {total} 对，RANSAC 内点 {inliers} 对")
    print("单应性矩阵 H:\n", np.round(H, 4))
    vis, boxes, dmap = diff_defects(ref, aligned)
    print("差分检出的缺陷框:", boxes)
    cv2.imwrite("docs/aligned_diff.png", vis)
```

**为什么必须 RANSAC**：`cv2.findHomography` 至少有 4 对点才能解，但匹配里必然混入错误对应（外点）。最小二乘会被一个外点带偏；RANSAC 反复随机抽样、用"内点最多"的模型做最终解，这是工业对齐的默认做法。

> 本节直接服务 P3：印刷品/装配完整性检测的标准链路就是 **对齐 → 差分 → 连通域筛选**（`:396`）。W11 只要把这里的 `make_pair()` 换成真图即可。
