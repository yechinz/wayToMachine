## 1. PyTorch 基础：张量、NCHW、训练循环（S-49）

```python
# src/algo/dl/train_toy.py —— 不下载任何数据，CPU 秒级跑完
import torch
import numpy as np

torch.manual_seed(0)
rng = np.random.default_rng(0)

# 合成二分类：两类二维点，故意加一点噪声
n = 400
X0 = rng.normal(loc=(-1.2, -0.8), scale=0.45, size=(n // 2, 2))
X1 = rng.normal(loc=(1.2, 0.9), scale=0.45, size=(n // 2, 2))
X = torch.tensor(np.vstack([X0, X1]), dtype=torch.float32)
y = torch.tensor([0] * (n // 2) + [1] * (n // 2), dtype=torch.long)

# NCHW：PyTorch 图像张量是 (Batch, Channel, Height, Width)
# 对比 OpenCV 的 HWC，所以 dnn.blobFromImage 要做转置 + 归一化
print("X.shape =", tuple(X.shape), "（这里是 N×2 特征，图像则是 N×3×H×W）")

model = torch.nn.Sequential(
    torch.nn.Linear(2, 16), torch.nn.ReLU(),
    torch.nn.Linear(16, 2))
opt = torch.optim.Adam(model.parameters(), lr=0.02)
loss_fn = torch.nn.CrossEntropyLoss()

for epoch in range(1, 121):
    opt.zero_grad()                 # ① 清梯度（Java 类比：每轮重置累加器）
    logits = model(X)               # ② 前向
    loss = loss_fn(logits, y)       # ③ 算损失
    loss.backward()                 # ④ 反向传播求梯度
    opt.step()                      # ⑤ 更新参数
    if epoch % 20 == 0:
        acc = (logits.argmax(1) == y).float().mean().item()
        print(f"epoch {epoch:3d}  loss={loss.item():.4f}  acc={acc:.3f}")

# 过拟合演示：参数多 + 数据少 + 训练久 → 训练 acc 高但泛化差
print("结论：torch 训练循环 = 前向 / loss / backward / step，四步固定不变")
```

**必须搞懂的概念**（`:218`）：张量（带 shape 的多维数组）、卷积（局部加权求和）、损失（预测与真值的差距）、反向传播（链式法则求梯度）、过拟合（记住了训练集的噪声）。

> `blobFromImage` 做的三件事（面试加分）：**HWC→CHW 转置、BGR→RGB 或减均值、缩放到网络输入尺寸并归一化**。张量形状对不上是部署第一大坑（S-55）。

```python
# src/algo/dl/voc_to_yolo.py —— VOC(xyxy 绝对像素) → YOLO(cxcywh 归一化)
from pathlib import Path
import xml.etree.ElementTree as ET

CLASSES = ["scratch", "dent", "stain"]


def convert(xml_path, img_w, img_h, out_dir):
    root = ET.parse(xml_path).getroot()
    lines = []
    for obj in root.findall("object"):
        name = obj.findtext("name")
        if name not in CLASSES:
            continue
        box = obj.find("bndbox")
        x1 = float(box.findtext("xmin")); x2 = float(box.findtext("xmax"))
        y1 = float(box.findtext("ymin")); y2 = float(box.findtext("ymax"))
        cx, cy = (x1 + x2) / 2 / img_w, (y1 + y2) / 2 / img_h
        w, h = (x2 - x1) / img_w, (y2 - y1) / img_h
        lines.append(f"{CLASSES.index(name)} {cx:.6f} {cy:.6f} {w:.6f} {h:.6f}")
    out = Path(out_dir) / (Path(xml_path).stem + ".txt")
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text("\n".join(lines), encoding="utf-8")
    return out


if __name__ == "__main__":
    demo_xml = Path("data/yolo/demo.xml")
    demo_xml.parent.mkdir(parents=True, exist_ok=True)
    demo_xml.write_text(
        "<annotation><object><name>scratch</name><bndbox>"
        "<xmin>100</xmin><ymin>80</ymin><xmax>180</xmax><ymax>140</ymax>"
        "</bndbox></object></annotation>", encoding="utf-8")
    print("生成:", convert(demo_xml, 640, 480, "data/yolo/labels"))
    print(Path("data/yolo/labels/demo.txt").read_text(encoding="utf-8"))
```

## 3. 分割、分类、小样本（S-51，W25–26）

| 任务 | 方案 | 标注成本 | 什么时候用 |
|---|---|---|---|
| 缺陷分类（整图 OK/NG） | YOLO-cls / ResNet 微调 | 最低（只打标签） | 只需判定，不需位置 |
| 缺陷检测（框） | YOLO11 / YOLO26 | 中（画框） | 要位置、要计数 |

| 小样本 few-shot | 传统特征 + SVM / 数据增强 / 异常检测 | 低 | NG 样本只有几十张 |

> **决策原则**：能用"整图分类"解决，就别做"检测"；能用"检测"解决，就别做"分割"。标注成本是项目成本的大头（`:444`）。

## 4. ★ 异常检测：PatchCore / EfficientAD（S-52，W27–28）

### 4.1 为什么工业最实用（`:445`、`:230`、加分题 `:838`）

- **只用良品训练**：产线 99% 是良品，NG 样本稀少且形态千变万化，"枚举所有缺陷"不现实。
- 思路：把良品图像喂进预训练 CNN 提特征，建一个"正常特征记忆库（memory bank）"；推理时算测试图特征到记忆库的最近邻距离，距离大 = 异常，输出异常热力图。
- 指标：**AUROC**（图像级/像素级），MVTec AD 是标准集（`:914`）。

### 4.2 数据集目录（MVTec AD，`:550`、`:914`）

```
data/mvtec/hazelnut/
├── train/good/          ← 只用这些训练（良品）
├── test/good/           ← 评估过杀率
├── test/crack/ ...      ← 评估漏检率
└── ground_truth/        ← 像素级标注（可选）
```

```python
# 概念骨架：理解"最近邻特征距离 = 异常分数"，不绑定具体库版本
import numpy as np


class TinyPatchCore:
    """教学用极简版：真实项目用 anomalib 的预训练特征提取器"""

    def fit(self, good_features):
        # good_features: (N, D)，N 张良品图的特征
        self.bank = np.asarray(good_features, np.float32)

    def score(self, feats):
        # feats: (M, D)；返回每个 patch 到记忆库的最小距离
        dist = np.linalg.norm(feats[:, None, :] - self.bank[None, :, :], axis=2)
        return float(dist.min(axis=1).max())     # 图像级分数取最异常 patch


if __name__ == "__main__":
    rng = np.random.default_rng(0)
    pc = TinyPatchCore()
    pc.fit(rng.normal(0, 0.1, (200, 16)))        # 良品特征
    print("良品分数:", round(pc.score(rng.normal(0, 0.1, (10, 16))), 4))
    print("异常分数:", round(pc.score(rng.normal(1.5, 0.1, (10, 16))), 4))
```

### 5.1 划分 + 增强（不依赖第三方增强库）

```python
# src/algo/dl/split_augment.py
import random
from pathlib import Path
import cv2
import numpy as np


def split_dataset(img_dir, ratio=(0.8, 0.1, 0.1), seed=42):
    files = sorted(Path(img_dir).glob("*.png")) + sorted(Path(img_dir).glob("*.jpg"))
    random.Random(seed).shuffle(files)
    a, b = int(len(files) * ratio[0]), int(len(files) * (ratio[0] + ratio[1]))
    return {"train": files[:a], "val": files[a:b], "test": files[b:]}


def augment(bgr, seed=0):
    rng = np.random.default_rng(seed)
    out = bgr.copy()
    if rng.random() < 0.5:
        out = cv2.flip(out, 1)                         # 左右翻转（注意对称性缺陷）
    if rng.random() < 0.5:
        out = cv2.convertScaleAbs(out, alpha=float(rng.uniform(0.8, 1.2)),
                                  beta=float(rng.uniform(-15, 15)))  # 模拟光照波动
    if rng.random() < 0.3:
        out = cv2.GaussianBlur(out, (3, 3), 0)         # 模拟轻微失焦
    return out


if __name__ == "__main__":
    parts = split_dataset("data/defects/ok")
    print({k: len(v) for k, v in parts.items()})
    print("增强输出尺寸:", augment(np.zeros((64, 64, 3), np.uint8)).shape)
```

> **类别不平衡**（`:222`）：NG 样本少时优先级是 ① 多采集/半自动标注 ② 针对 NG 做过采样与增强 ③ 改用异常检测（只用良品）④ 最后才调 loss 权重。**不要用"复制粘贴 NG 图凑数"骗自己。**

### 5.2 指标与阈值扫描（本章最实用的一段代码）

```python
# src/algo/dl/metrics.py
import numpy as np


def sweep_threshold(scores, labels, steps=101):
    """scores 越大越像 NG；labels=1 表示真 NG
    输出：漏检率(escape) / 过杀率(false_reject) 随阈值的变化"""
    scores = np.asarray(scores, float)
    labels = np.asarray(labels, int)
    rows = []
    for th in np.linspace(scores.min(), scores.max(), steps):
        pred = (scores >= th).astype(int)
        tp = int(((pred == 1) & (labels == 1)).sum())
        fn = int(((pred == 0) & (labels == 1)).sum())
        fp = int(((pred == 1) & (labels == 0)).sum())
        tn = int(((pred == 0) & (labels == 0)).sum())
        escape = fn / max(tp + fn, 1)          # 漏检率：不良品被判 OK
        overkill = fp / max(fp + tn, 1)        # 过杀率：良品被判 NG
        rows.append((round(float(th), 4), round(escape, 4), round(overkill, 4)))
    return rows


if __name__ == "__main__":
    rng = np.random.default_rng(1)
    good = rng.normal(0.2, 0.12, 1000)
    bad = rng.normal(0.7, 0.15, 100)
    scores = np.concatenate([good, bad])
    labels = np.concatenate([np.zeros(1000), np.ones(100)])
    rows = sweep_threshold(scores, labels)
    # 找"漏检率 < 1%"前提下过杀率最低的阈值
    ok = [r for r in rows if r[1] < 0.01]
    print("满足漏检<1% 的阈值里，过杀最低:", min(ok, key=lambda r: r[2]))
    print("阈值提高 → 漏检升高、过杀降低；两者此消彼长（:452）")
```

| 指标 | 定义 | 产线后果 |
|---|---|---|
| **漏检率 Escape** | 不良品判 OK | 客诉、召回，**老板骂你** |
| **过杀率 False Reject** | 良品判 NG | 停线、产能损失，**工人骂你，甚至把系统关掉** |
| 召回率 | 1 − 漏检率 | 工业常要求 > 99% |
| mAP | 检测任务综合指标 | 关注定位+分类整体质量 |

> **面试标准答案（`:450-453`、`:836`）**：两者此消彼长；**产线更怕漏检**（流出不良 = 客诉召回），但过杀率过高会让系统被现场弃用；**阈值最终是商务问题**，必须和客户在验收标准里写清楚（例如"召回 > 99%、过杀 < 1%"）。

### 6.1 导出 ONNX + CPU 推理

```python
# src/algo/dl/export_infer.py
import time
import numpy as np
import torch
import cv2


def export_dummy_model(path="configs/toy.onnx"):
    """用一个小 CNN 演示导出；真实项目换成训练好的模型"""
    model = torch.nn.Sequential(
        torch.nn.Conv2d(3, 8, 3, padding=1), torch.nn.ReLU(),
        torch.nn.AdaptiveAvgPool2d((1, 1)), torch.nn.Flatten(),
        torch.nn.Linear(8, 2))
    model.eval()
    dummy = torch.randn(1, 3, 64, 64)
    torch.onnx.export(model, dummy, path,
                      input_names=["input"], output_names=["logits"],
                      opset_version=12, dynamic_axes={"input": {0: "batch"}})
    print("已导出:", path)
    return path


def infer_onnx(path, bgr, size=64):
    try:
        import onnxruntime as ort
    except ImportError:
        raise SystemExit("先安装：pip install onnxruntime")
    sess = ort.InferenceSession(path, providers=["CPUExecutionProvider"])
    print("输入名/形状:", [(i.name, i.shape) for i in sess.get_inputs()])
    # OpenCV HWC/BGR → NCHW/RGB + 归一化（blobFromImage 等价手写）
    rgb = cv2.cvtColor(cv2.resize(bgr, (size, size)), cv2.COLOR_BGR2RGB)
    x = rgb.transpose(2, 0, 1)[None].astype(np.float32) / 255.0
    t0 = time.perf_counter()
    out = sess.run(None, {"input": x})[0]
    ms = (time.perf_counter() - t0) * 1000
    print(f"单帧推理 {ms:.2f} ms，logits = {np.round(out[0], 3)}")
    return out, ms


if __name__ == "__main__":
    p = export_dummy_model()
    infer_onnx(p, np.full((480, 640, 3), 128, np.uint8))
```

```csharp
using System;
using System.Linq;
using System.Runtime.InteropServices;   // Marshal.Copy
using OpenCvSharp;
using Microsoft.ML.OnnxRuntime;
using Microsoft.ML.OnnxRuntime.Tensors;

using var mat = Cv2.ImRead("data/raw/test.png", ImreadModes.Color);
Cv2.Resize(mat, mat, new Size(64, 64));
Cv2.CvtColor(mat, mat, ColorConversionCodes.BGR2RGB);

// HWC → CHW，float32 / 255
var data = new float[1 * 3 * 64 * 64];
var idx = new byte[3 * 64 * 64];
Marshal.Copy(mat.Data, idx, 0, idx.Length);      // 连续 BGR→已转 RGB 的字节
for (int c = 0; c < 3; c++)
    for (int y = 0; y < 64; y++)
        for (int x = 0; x < 64; x++)
            data[c * 64 * 64 + y * 64 + x] = idx[(y * 64 + x) * 3 + c] / 255f;

var tensor = new DenseTensor<float>(data, new[] { 1, 3, 64, 64 });
using var session = new InferenceSession("configs/toy.onnx");
using var results = session.Run(new[] { NamedOnnxValue.CreateFromTensor("input", tensor) });
var logits = results.First().AsTensor<float>().ToArray();
Console.WriteLine($"logits = {string.Join(", ", logits)}");
```

### 6.3 无 GPU 的性能优化顺序

| 手段 | 效果 | 代价 |
|---|---|---|
| 降输入分辨率 640→320 | 约 4× | 小缺陷可能漏 |
| OpenVINO / ORT 优化 + 多线程 | 2–4× | 需按 CPU 型号调 |
| FP16 / INT8 量化 | 2–4× | 需校准集，精度略降 |
| ROI 裁切后推理 | 大幅 | 需先定位 |
| SAHI 切片推理（大图小缺陷，`:447`） | 保精度 | 帧数变多 |

> 验收"单帧 < 50ms"（`:228`）在 CPU 上达不到时，报告里写清：`输入尺寸 / 硬件型号 / 线程数 / 实测 P95 / 优化前后对比`。**工程诚信比好看的数字重要。**

## 7. P5 字符识别收口与传统 vs DL 决策树

**P5**（`:586-598`）：200 张喷码/铭牌 → PaddleOCR → 固定字符集小 CNN → 业务校验（格式/日期/MES 比对）。代码见 [`07` 号手册 §7](./07_学习手册_阶段2_成像打光缺陷检测与定位识别.md)。
**认知**：难点在成像与校验逻辑，不在识别模型本身。

```text
拿到缺陷检测需求
├─ 缺陷形态可枚举 + 对比度高 + 良品纹理简单 → 传统（Blob/差分/形态学），1ms、可解释
├─ 只需整图 OK/NG 且 NG 样本极少         → 异常检测 PatchCore（只用良品）
├─ 需要框出位置且 NG 样本充足             → YOLO 检测
├─ 需要像素级面积/边界                    → 分割（先评估标注成本）
└─ 大图小缺陷                             → 切片推理 SAHI / 先 ROI 定位
```
