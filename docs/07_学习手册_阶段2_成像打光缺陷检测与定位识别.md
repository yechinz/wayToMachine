## 1. 成像硬件与选型计算（S-39、S-40、S-41）

### 1.1 相机参数速查（`:257-264`）

| 参数 | 要点 | 选型判断 |
|---|---|---|
| 分辨率 | 决定精度上限，500 万（2448×2048）入门标配 | 先算 `FOV/像素数`，再乘 3–5 像素作为稳定精度 |
| 快门 | 全局快门拍运动物体，卷帘会果冻变形 | 皮带/飞拍 → 必须全局 |
| 帧率 | 决定节拍上限，注意"全分辨率"还是"ROI 帧率" | 别被宣传页的 ROI 帧率骗了 |
| 接口 | USB3（≤5m）、GigE（100m，工业主流）、Camera Link/CoaXPress | 产线优先 GigE |
| 黑白 / 彩色 | **测量与缺陷优先黑白**（灵敏度高约 3 倍、无插值伪影） | 只有要区分颜色才上彩色 |
| 靶面 | 1/1.8"、2/3"、1"… 必须和镜头匹配 | 小靶面镜头配大靶面 → 暗角 |

### 1.2 镜头三公式 + 景深 + 曝光（`:268-292`、`:324-328`）

```
焦距 f    = 工作距离 WD × 靶面尺寸 / 视场 FOV
放大倍率 β = 靶面尺寸 / FOV
单像素精度 = FOV宽度 / 相机水平像素数
景深 DOF  ≈ 2 · N · c · (1+β) / β²        N=光圈F数，c=允许弥散圆（取 2 倍像元）
曝光上限 t ≤ 允许模糊量 / 运动速度
```

```python
# src/algo/selection.py —— 选型计算器，可直接改成命令行小工具
import math


def lens_calc(wd_mm, sensor_w_mm, fov_w_mm, pixels_w,
              fnum=8.0, pixel_um=3.45, speed_mm_s=0.0, blur_allow_mm=0.1):
    f = wd_mm * sensor_w_mm / fov_w_mm
    beta = sensor_w_mm / fov_w_mm
    mm_per_px = fov_w_mm / pixels_w
    c_mm = 2 * pixel_um / 1000.0
    dof = 2 * fnum * c_mm * (1 + beta) / (beta ** 2)
    t_ms = (blur_allow_mm / speed_mm_s * 1000.0) if speed_mm_s > 0 else None
    print(f"焦距 f        = {f:.2f} mm   → 选接近的标准焦距")
    print(f"放大倍率 β    = {beta:.4f}")
    print(f"单像素精度    = {mm_per_px:.5f} mm/px")
    print(f"稳定测量精度  = {mm_per_px*3:.3f} ~ {mm_per_px*5:.3f} mm（取 3–5 px）")
    print(f"景深 DOF      = {dof:.1f} mm（倍率越大越浅）")
    if t_ms is not None:
        print(f"曝光上限      = {t_ms:.3f} ms → 需强光短曝光 / 频闪 + 硬触发")
    return dict(f=f, beta=beta, mm_per_px=mm_per_px, dof=dof, exposure_ms=t_ms)


if __name__ == "__main__":
    # 路线图 :276-282 的例题
    lens_calc(wd_mm=300, sensor_w_mm=7.2, fov_w_mm=100, pixels_w=2448,
```

```python
# src/capture/lighting_log.py —— 用直方图量化"对比度"，别只靠肉眼
import cv2
import numpy as np


def contrast_score(gray, roi=None):
    """ROI 内目标/背景分离度：用 Otsu 阈值后的类间方差做粗略评分"""
    if roi is not None:
        x, y, w, h = roi
        gray = gray[y:y + h, x:x + w]
    t, _ = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    a, b = gray[gray <= t], gray[gray > t]
    if len(a) == 0 or len(b) == 0:
        return 0.0
    w0, w1 = len(a) / gray.size, len(b) / gray.size
    return float(w0 * w1 * (a.mean() - b.mean()) ** 2)


if __name__ == "__main__":
    # 每次实验填一行：编号 / 光源 / 角度 / 曝光 / 评分 / 结论
    rows = [
        ("L01", "环形高角度 0°",       "1/500", None),
        ("L02", "环形低角度 15°",      "1/500", None),
        ("L03", "背光",                "1/250", None),
        ("L04", "同轴（模拟）",         "1/250", None),
        ("L05", "条形侧光 30°",         "1/500", None),
    ]
    for no, light, exp, _ in rows:
        print(f"| {no} | {light} | {exp} | 填入 contrast_score | 填入结论 |")
    print("20 组全做完，再决定算法路线 —— 这是路线图 :198 的硬性要求")
```

## 2. 四大任务方法论总纲（S-42，`:361-413`）

| 任务 | 输出 | 传统方法 | 何时必须上 DL |
|---|---|---|---|
| 定位 Where | `(x, y, θ)` 像素 | 模板匹配 / Blob 重心 / 特征点+单应性 | 形变大、纹理复杂、遮挡 |
| 测量 How big | mm | 标定 + 亚像素 + 拟合 | 一般不用 DL |
| 检测 Good/Bad | OK/NG + 缺陷框 | Blob / 差分 / 形态学 / 频域 / 光度立体 | 缺陷不可枚举、对比度极低、良品纹理复杂、需分类 |
| 识别 What | 字符串 / 类别 | 读码 / OCR / 颜色统计 | 通用字符、复杂背景 |

**"什么时候传统方法做不了"四判据（原文 `:402-406`）**：
① 缺陷形态不可枚举（划痕千变万化）；② 缺陷与背景对比度极低；③ 良品本身纹理复杂（皮革/木材/铸件）；④ 需要分类（划痕还是脏污）。

## 3. 传统缺陷检测六法（S-43，W11–12）

### 3.1 法一 · Blob 分析（工业最常用，`:395`、`:669-698`）

```python
# src/algo/blob_defect.py
import cv2
import numpy as np


def detect_defects(img, min_area=40, max_aspect=6.0, min_fill=0.15):
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    blur = cv2.GaussianBlur(gray, (5, 5), 0)
    # 暗缺陷 → THRESH_BINARY_INV；亮缺陷去掉 _INV
    _, bw = cv2.threshold(blur, 0, 255,
                          cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    bw = cv2.morphologyEx(bw, cv2.MORPH_OPEN, k)
    bw = cv2.morphologyEx(bw, cv2.MORPH_CLOSE, k)

    n, labels, stats, cents = cv2.connectedComponentsWithStats(bw, 8)
    out, results = img.copy(), []
    for i in range(1, n):                       # 0 是背景
        x, y, w, h, area = stats[i]
        if area < min_area:
            continue
        aspect = max(w, h) / max(min(w, h), 1)
        fill = area / float(w * h)
        if aspect > max_aspect or fill < min_fill:
            continue
        results.append(dict(box=(int(x), int(y), int(w), int(h)),
                            area=int(area), aspect=round(float(aspect), 2),
                            fill=round(float(fill), 2),
                            center=tuple(np.round(cents[i], 1))))
        cv2.rectangle(out, (x, y), (x + w, y + h), (0, 0, 255), 2)
    return out, results


if __name__ == "__main__":
    img = np.full((300, 400, 3), 190, np.uint8)
    cv2.circle(img, (100, 150), 12, 45, -1)      # 暗脏点
    cv2.line(img, (200, 100), (280, 100), 50, 3)  # 暗划痕
    cv2.circle(img, (330, 220), 4, 255, -1)       # 亮点干扰
    vis, res = detect_defects(img)
    print("检出:", res)
```

> **调参顺序**：先保证缺陷被切出来（阈值）→ 再去噪（开运算）→ 最后用面积/长宽比/填充率筛。**顺序反了会把缺陷一起滤掉。**

### 3.4 法四 · 频域陷波去周期性纹理（`:398`）

织物、网格、拉丝金属有强周期性纹理，空域滤波很难去掉；在频域把对应的亮点对（±频率）挖掉即可。

```python
# src/algo/freq_notch.py
import cv2
import numpy as np


def make_textured_defect():
    """合成：横向周期纹理 + 一处暗缺陷"""
    h, w = 256, 256
    yy, xx = np.mgrid[0:h, 0:w]
    tex = (128 + 35 * np.sin(2 * np.pi * xx / 12)).astype(np.uint8)
    cv2.circle(tex, (170, 130), 10, 40, -1)
    rng = np.random.default_rng(0)
    return np.clip(tex.astype(np.float32) + rng.normal(0, 3, tex.shape),
                   0, 255).astype(np.uint8)


def notch_filter(gray, peaks, radius=4):
    """peaks: [(dy, dx)] 相对频谱中心的周期峰偏移（±两个都要挖）"""
    dft = cv2.dft(np.float32(gray), flags=cv2.DFT_COMPLEX_OUTPUT)
    shifted = np.fft.fftshift(dft, axes=(0, 1))
    mask = np.ones(gray.shape, np.float32)
    cy, cx = gray.shape[0] // 2, gray.shape[1] // 2
    for dy, dx in peaks:
        for sy, sx in ((dy, dx), (-dy, -dx)):
            cv2.circle(mask, (cx + sx, cy + sy), radius, 0.0, -1)
    shifted[:, :, 0] *= mask
    shifted[:, :, 1] *= mask
    inv = np.fft.ifftshift(shifted, axes=(0, 1))
    back = cv2.idft(inv, flags=cv2.DFT_SCALE | cv2.DFT_REAL_OUTPUT)
    return cv2.normalize(back, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)


if __name__ == "__main__":
    img = make_textured_defect()
    # 横向周期 12px → 频谱上约在 (±21, 0) 处（256/12≈21.3）
    cleaned = notch_filter(img, [(0, 21)], radius=4)
    print("原图标准差:", round(float(img.std()), 2),
          " 陷波后:", round(float(cleaned.std()), 2))
    _, bw = cv2.threshold(cleaned, 0, 255,
                          cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    n, labels, stats, cents = cv2.connectedComponentsWithStats(bw, 8)
    print("去纹理后连通域数:", n - 1, " 最大块面积:",
          int(stats[1:, cv2.CC_STAT_AREA].max()) if n > 1 else 0)
```

### 3.5 法五 · 光度立体（高级但有效，`:399`）

原理：`I = ρ · (L · n)`，用 ≥3 个已知方向的光源各拍一张，逐像素解出表面法向 `n`，凹坑/压伤会在法向梯度图上暴露。**前提**：光源方向标定过、相机响应线性、工件静止。

```python
# 简化示意：3 个方向光各拍一张 → 逐像素最小二乘求法向
import numpy as np


def photometric_stereo(images, light_dirs):
    """images: 三张同视角、不同光源方向的灰度图 list[h,w]
       light_dirs: 3x3 归一化光方向矩阵（需事先标定）
       返回：法向图 (h,w,3) 与凹坑响应图 (h,w)"""
    L = np.asarray(light_dirs, np.float32)
    stack = np.stack(images, axis=-1).astype(np.float32) / 255.0   # (h,w,3)
    normals = np.linalg.solve(L, stack)        # numpy 广播求解，输出 (h,w,3)
    gx = np.gradient(normals[:, :, 0], axis=1)
    gy = np.gradient(normals[:, :, 1], axis=0)
    return normals, np.hypot(gx, gy)          # 梯度大处 = 法向突变 = 凹坑/压伤


if __name__ == "__main__":
    # 演示：三张平坦表面的合成光照图（实际项目换成真实多光源采集）
    h = w = 64
    L = np.array([[0.0, 0.0, 1.0], [0.6, 0.0, 0.8], [0.0, 0.6, 0.8]], np.float32)
    L /= np.linalg.norm(L, axis=1, keepdims=True)
    flat = np.full((h, w), 0.5, np.float32)
    imgs = [flat * L[i, 2] * 255 for i in range(3)]
    n, dent = photometric_stereo(imgs, L)
    print("法向图形状:", n.shape, " 凹坑响应均值:", round(float(dent.mean()), 6))
```

> `images` 必须由真实多光源采集提供（光源方向、相机线性响应、平场校正都要先做）；这里只给算法骨架，实做时先在标准件上验证法向是否合理。

## 5. 九点标定与坐标链路（S-45、S-46，W15–16）

### 5.1 为什么需要它

模板匹配给的是**像素坐标**，机器人要的是**机械坐标（mm）**。九点标定求一个 2×3 仿射矩阵，把两者对上（`:423`）。

### 5.2 完整代码（在路线图 `:700-723` 基础上加残差与噪声实验）

```python
# src/algo/nine_point.py
import cv2
import numpy as np
import json
from pathlib import Path


def calibrate_nine(pix, rob, out="configs/nine_point.json"):
    M, inliers = cv2.estimateAffine2D(np.float32(pix), np.float32(rob))
    if M is None:
        raise RuntimeError("仿射求解失败，检查点位是否共线")
    pred = cv2.transform(np.float32(pix).reshape(-1, 1, 2), M).reshape(-1, 2)
    err = np.linalg.norm(pred - np.float32(rob), axis=1)
    print("仿射矩阵 M =\n", np.round(M, 6))
    print(f"最大残差 {err.max():.3f} mm，平均 {err.mean():.3f} mm，"
          f"内点数 {int(inliers.sum()) if inliers is not None else 'N/A'}")
    if err.max() > 0.1:
        print("⚠️ 残差 > 0.1mm：重新采点或先做畸变校正（:431、:725）")
    Path(out).parent.mkdir(parents=True, exist_ok=True)
    Path(out).write_text(json.dumps({"M": M.tolist(),
                                    "max_residual_mm": float(err.max()),
                                    "mean_residual_mm": float(err.mean())},
                                   ensure_ascii=False, indent=2),
                         encoding="utf-8")
    return M, err


def to_robot(M, u, v):
    return cv2.transform(np.float32([[[u, v]]]), M).reshape(2)


if __name__ == "__main__":
    # 9 点覆盖整个视野，不要挤在中间（:429）
    pix = np.array([[120, 80], [640, 75], [1150, 82],
                    [125, 480], [635, 485], [1145, 478],
                    [118, 880], [642, 885], [1152, 878]], np.float32)
    rob = np.array([[0, 0], [50, 0], [100, 0],
                    [0, 40], [50, 40], [100, 40],
                    [0, 80], [50, 80], [100, 80]], np.float32)
    M, err = calibrate_nine(pix, rob)
    print("像素 (640, 480) → 机械", np.round(to_robot(M, 640, 480), 3), "mm")

    # 破坏性实验：点位全挤在中心 → 外推误差被放大
    center_pix = (pix - pix.mean(axis=0)) * 0.2 + pix.mean(axis=0)
    _, err2 = calibrate_nine(center_pix, rob, out="configs/nine_point_bad.json")
    print("挤在中心的平均残差:", round(float(err2.mean()), 4))
```

### 5.3 坐标链路与每一步误差来源（阶段 2 验收 `:207`）

```
相机成像 → 畸变校正(remap) → 定位(u,v,θ) → 九点仿射 → 机械坐标(X,Y) → 机器人/PLC
   ↑镜头畸变      ↑内参误差      ↑定位±0.3px   ↑采点/拟合/剪切  ↑机械重复定位/高度变化
```

| 误差来源 | 对策 |
|---|---|
| 镜头畸变（边缘最大） | 先 `remap` 校正，再采九点 |
| 九点只覆盖中心 | 覆盖全视野四角（`:429`） |
| 采点随机误差 | 每点采 3–5 次取平均（`:430`） |
| 工件 Z 高度变化 | 仿射解决不了，用远心镜头/固定夹具（`:432`） |
| 机器人本体重复定位精度 | 验收时要单独测，别全赖视觉 |

### 5.4 eye-in-hand vs eye-to-hand（面试题 13 `:829`）

| | 相机装在哪 | 求什么 | 典型场景 |
|---|---|---|---|
| eye-in-hand | 机器人末端，跟着手走 | 相机 ↔ 末端法兰 | 抓取后二次定位 |
| eye-to-hand | 固定在机架上 | 相机 ↔ 机器人基座 | 上料定位（最常见） |

两者都用 `cv2.calibrateHandEye`，输入多组"机器人姿态 + 标定板姿态"。**2D 场景（平面工件、高度固定）九点仿射就够了，不要一上来上手眼 3D 标定。**

## 6. P4 定位抓取 + PLC 握手（W15–16，`:567-582`）

```python
# src/comm/plc_stub.py —— 不依赖 pymodbus 版本的最小 TCP 从站（模拟 PLC）
# 路线图 :473 提醒 pymodbus 3.x API 有变更；先用 socket 把"握手逻辑"跑通
import socket
import threading
import time


class FakePLC:
    """极简协议：客户端发 'TRIG\\n' → 回 'ACK\\n'；发 'RESULT,OK\\n' → 回 'ACK\\n'"""

    def __init__(self, host="127.0.0.1", port=8502):
        self.srv = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.srv.bind((host, port))
        self.srv.listen(1)
        self.alive = True
        self.thread = threading.Thread(target=self._loop, daemon=True)

    def _loop(self):
        while self.alive:
            try:
                conn, _ = self.srv.accept()
            except OSError:
                break
            with conn:
                conn.settimeout(2.0)
                try:
                    data = conn.recv(1024).decode("utf-8", "ignore").strip()
                except socket.timeout:
                    continue
                if data:
                    print("[PLC] 收到:", data)
                    conn.sendall(b"ACK\n")
        self.srv.close()

    def start(self):
        self.thread.start()

    def stop(self):
        self.alive = False
        self.srv.close()


def request_once(payload, host="127.0.0.1", port=8502, timeout=2.0):
    """带超时的单次请求：超时必须判 NG，绝不能卡死（:483）"""
    try:
        with socket.create_connection((host, port), timeout=timeout) as s:
            s.sendall((payload + "\n").encode("utf-8"))
            return s.recv(1024).decode("utf-8", "ignore").strip()
    except OSError as e:
        return f"TIMEOUT:{e}"


if __name__ == "__main__":
    plc = FakePLC()
    plc.start()
    time.sleep(0.1)
    print("触发:", request_once("TRIG"))
    print("回结果:", request_once("RESULT,OK"))
    print("模拟断线:", request_once("TRIG", port=8599))
    plc.stop()
```

**握手状态机（禁止裸 `sleep`，`:482`）**：

```
IDLE → (收到 TRIG) → CAPTURE → ALGO → SEND_RESULT → WAIT_ACK → IDLE
任意状态超时 / 异常 → NG → ALARM → 自动重连 → IDLE
```

> 生产代码请把上面的 socket 换成 `pymodbus`（3.x 用 `ModbusTcpClient` + `read_holding_registers` 去掉 `unit=` 参数，改 `slave=`）、S7 用 `python-snap7`、OPC UA 用 `asyncua`（`:469-476`）。**协议可以换，状态机与超时设计不能换。**

**P4 验收**（`:578-580`）：20 个随机位置定位误差 < 0.3mm、角度 < 0.5°；能解释边缘误差更大；通信层断线重连后能继续。

## 7. 识别类：码、OCR、颜色、业务校验（S-47，W17–18）

```python
# src/algo/recognize.py
import re
from datetime import datetime
import cv2
import numpy as np


def read_qr(bgr):
    det = cv2.QRCodeDetector()
    data, pts, _ = det.detectAndDecode(bgr)
    return data, pts


def ocr_paddle(bgr):
    """需额外安装：pip install paddleocr paddlepaddle（CPU 版即可）"""
    try:
        from paddleocr import PaddleOCR
    except ImportError:
        return "[未安装 paddleocr，先跑通传统链路]"
    ocr = PaddleOCR(use_angle_cls=True, lang="ch", show_log=False)
    result = ocr.ocr(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    texts = [line[1][0] for block in result if block for line in block]
    return "".join(texts)


def business_check(text):
    """工业 OCR 真正的价值在这里：格式 + 合法性 + 与 MES 比对（:594、:598）"""
    text = text.strip().upper()
    if not re.fullmatch(r"LOT\d{4}-[A-Z]\d{3}", text):
        return False, "批号格式不符"
    m = re.search(r"(\d{4})(\d{2})(\d{2})", text)
    if m:
        try:
            datetime.strptime("".join(m.groups()), "%Y%m%d")
        except ValueError:
            return False, "日期不合法"
    return True, "OK"


def color_classify(bgr):
    hsv = cv2.cvtColor(bgr, cv2.COLOR_BGR2HSV)
    h, s, v = cv2.split(hsv)
    if float(s.mean()) < 30:
        return "灰/白"
    hv = float(h.mean())
    if hv < 10 or hv > 170:
        return "红"
    if hv < 35:
        return "橙/黄"
    if hv < 85:
        return "绿"
    return "蓝"


if __name__ == "__main__":
    print(business_check("LOT2026-A001"))
    print(business_check("LOT2026-A00"))       # 故意失败
    img = np.full((80, 200, 3), (30, 30, 200), np.uint8)
    print("颜色分类:", color_classify(img))
    print("二维码:", read_qr(img))
```

> **关键认知（`:598`）**：工业 OCR 的难点不在识别，在**成像**（打光让字符凸显）与**校验逻辑**（识别对但内容非法照样 NG）。DPM 码建议直接用 Cognex/Keyence 读码器（`:410`），别硬啃 OpenCV。
