### 1.4 版本兼容写法（贯穿全套手册）

OpenCV 3.x 返回 3 个值、4.x/5.x 返回 2 个值的典型函数：`findContours`。**本套手册统一用 `[-2]` 取轮廓**，一份代码通吃：

```python
cnts = cv2.findContours(bw, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)[-2]
# 等价于 4.x: contours, hierarchy = cv2.findContours(...)
# 等价于 3.x: _, contours, hierarchy = cv2.findContours(...)
```

旧版 4.x 或 `headless` 包可能没有 `SIFT`；**5.0 主包已自带**（实测 `hasattr(cv2,"SIFT_create") == True`）。下面保留兼容写法，是为了让你读懂网上的 4.x 代码：

```python
def make_feature():
    if hasattr(cv2, "SIFT_create"):
        return cv2.SIFT_create(), cv2.NORM_L2
    return cv2.ORB_create(nfeatures=800), cv2.NORM_HAMMING  # 主包也有 ORB
```

## 3. 图像 IO 与像素地基（S-02、S-03）

### 3.1 读写的四个事实

| 事实 | 说明 | 踩坑后果 |
|---|---|---|
| `imread` 默认读成 **BGR 三通道 uint8** | 不是 RGB | 用 matplotlib 直接 `plt.imshow(img)` 会看到颜色发蓝 |
| 读失败返回 `None`，**不抛异常** | 路径含中文/不存在/格式不支持 | 下一步 `cvtColor` 报 `!_src.empty()`，新人查半天 |
| `imshow` 必须配 `waitKey` | 窗口刷新靠消息循环 | 窗口一闪而过 / 假死 |
| `uint8` 加减会**回绕饱和** | `250 + 10 = 4`（溢出）或按 OpenCV 饱和到 255 | 亮度增强出现黑斑 |

### 3.2 BGR / 通道 / 位深

```python
import cv2
import numpy as np

img = np.zeros((100, 160, 3), np.uint8)
img[:] = (30, 120, 200)                 # B=30, G=120, R=200（偏橙）
b, g, r = cv2.split(img)                # 三个单通道 uint8
print("shape:", img.shape, "B 均值:", b.mean(), "R 均值:", r.mean())

hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
print("HSV:", hsv[50, 80].tolist(), "灰度:", int(gray[50, 80]))

# 位深：uint8(0-255) ↔ float32(0-1)，做数学运算前常转浮点
f = img.astype(np.float32) / 255.0
print("float 范围:", f.min(), f.max(), "dtype:", f.dtype)
```

### 3.3 ROI 是视图，不是拷贝（本章最重要的一条）

```python
img = np.zeros((100, 100, 3), np.uint8)
roi = img[20:60, 30:80]        # 切片 = 视图（共享内存）
roi[:] = (0, 0, 255)           # 原图被改了！
print("原图中心像素:", img[40, 50].tolist())   # [0, 0, 255]

copy = img[20:60, 30:80].copy()   # 需要独立副本时必须 .copy()
copy[:] = (255, 0, 0)
print("原图仍是:", img[40, 50].tolist())      # [0, 0, 255] 未变
```

> **Java 类比**：切片视图 ≈ `ByteBuffer.duplicate().position()/limit()` 后的子缓冲，改的是同一块堆外内存；`.copy()` ≈ `Arrays.copyOfRange()`。

### 3.4 中文路径读写（Windows 必会）

```python
import cv2
import numpy as np

def imwrite_unicode(path, img, ext=".png"):
    ok, buf = cv2.imencode(ext, img)
    assert ok, "编码失败"
    with open(path, "wb") as f:
        f.write(buf.tobytes())

def imread_unicode(path, flag=cv2.IMREAD_COLOR):
    raw = np.fromfile(path, dtype=np.uint8)
    img = cv2.imdecode(raw, flag)
    assert img is not None, f"读图失败: {path}"
    return img

img = np.full((50, 50, 3), 200, np.uint8)
```

## 4. 第一个工具：看图器（T0-02 / W1 的 T1-01 提前做）

**功能**：鼠标移到哪，就显示那点的 BGR / HSV / 灰度值；按键 `g` 切灰度、`h` 看直方图、`s` 保存当前视图、`q` 退出。
**为什么先做它**：后面 8 周所有"这个阈值该取多少"的判断，都靠这个工具先看像素值。

```python
# src/ui/image_viewer.py —— 无相机也能跑（自动生成合成图）
import argparse
import cv2
import numpy as np


def make_demo():
    """合成一张有渐变、圆、文字的图，方便观察像素值"""
    h, w = 360, 640
    img = np.zeros((h, w, 3), np.uint8)
    for x in range(w):                       # 横向亮度渐变
        img[:, x] = (int(x / w * 255), 80, 160)
    cv2.circle(img, (200, 180), 70, (0, 255, 255), -1)
    cv2.rectangle(img, (400, 90), (560, 260), (255, 0, 0), -1)
    cv2.putText(img, "vision-lab", (40, 330),
                cv2.FONT_HERSHEY_SIMPLEX, 1.0, (255, 255, 255), 2)
    return img


class Viewer:
    def __init__(self, img):
        self.orig = img
        self.show_gray = False
        self.pos = (0, 0)

    def current(self):
        return (cv2.cvtColor(self.orig, cv2.COLOR_BGR2GRAY)
                if self.show_gray else self.orig)

    def on_mouse(self, event, x, y, flags, param):
        if event == cv2.EVENT_MOUSEMOVE:
            self.pos = (x, y)

    def render(self):
        cur = self.current()
        vis = cv2.cvtColor(cur, cv2.COLOR_GRAY2BGR) if cur.ndim == 2 else cur.copy()
        x, y = self.pos
        if 0 <= y < vis.shape[0] and 0 <= x < vis.shape[1]:
            b, g, r = self.orig[y, x].tolist()
            hsv = cv2.cvtColor(self.orig[y:y + 1, x:x + 1],
                               cv2.COLOR_BGR2HSV)[0, 0].tolist()
            gray = int(cv2.cvtColor(self.orig[y:y + 1, x:x + 1],
                                    cv2.COLOR_BGR2GRAY)[0, 0])
            txt = f"({x},{y}) BGR=({b},{g},{r}) HSV={tuple(hsv)} GRAY={gray}"
            cv2.rectangle(vis, (0, 0), (vis.shape[1], 28), (0, 0, 0), -1)
            cv2.putText(vis, txt, (8, 20), cv2.FONT_HERSHEY_SIMPLEX,
                        0.55, (255, 255, 255), 1, cv2.LINE_AA)
            cv2.drawMarker(vis, (x, y), (0, 255, 0), cv2.MARKER_CROSS, 12, 1)
        cv2.imshow("viewer", vis)

    def histogram(self):
        gray = cv2.cvtColor(self.orig, cv2.COLOR_BGR2GRAY)
        hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
        canvas = np.zeros((200, 256, 3), np.uint8)
        cv2.normalize(hist, hist, 0, 180, cv2.NORM_MINMAX)
        for i, v in enumerate(hist.ravel()):
            cv2.line(canvas, (i, 199), (i, 199 - int(v)), (0, 255, 0), 1)
        cv2.imshow("histogram", canvas)

    def run(self):
        cv2.namedWindow("viewer")
        cv2.setMouseCallback("viewer", self.on_mouse)
        print("g=灰度  h=直方图  s=保存  q=退出")
        while True:
            self.render()
            key = cv2.waitKey(30) & 0xFF
            if key == ord("q"):
                break
            elif key == ord("g"):
                self.show_gray = not self.show_gray
            elif key == ord("h"):
                self.histogram()
            elif key == ord("s"):
                cv2.imwrite("docs/viewer_snapshot.png", self.current())
                print("已保存 docs/viewer_snapshot.png")
        cv2.destroyAllWindows()


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--img", default=None, help="不传则使用合成图")
    args = ap.parse_args()
    src = cv2.imread(args.img) if args.img else make_demo()
    assert src is not None, "图片读取失败，检查路径/中文路径问题"
    Viewer(src).run()
```
